**参考代码**:

```
uint64 sys_pipe(uint64 fdarray) {
    struct proc *p = curr_proc();
    struct file* f0 = filealloc();
    struct file* f1 = filealloc();
    pipealloc(f0, f1);
    fd0 = fdalloc(f0);
    fd1 = fdalloc(f1);
    copyout(p->pagetable, fdarray, &fd0, sizeof(fd0));
    copyout(p->pagetable, fdarray + sizeof(uint64), &fd1, sizeof(fd1));
    return 0;
}
```
