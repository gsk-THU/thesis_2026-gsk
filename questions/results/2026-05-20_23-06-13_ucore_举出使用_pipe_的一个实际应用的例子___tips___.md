int pipealloc(struct file *f0, struct file *f1)
{
    struct pipe *pi = (struct pipe*)kalloc();
    if(pi == 0) return -1;  // 假设的内存分配失败检查
    pi->readopen = 1;
    pi->writeopen = 1;
    pi->nwrite = 0;

    f0->writable = 0;

3. 如果系统中没有 `dup2` 系统调用，你如何通过 `close` 和 `dup` 的组合实现同样的重定向效果？写出关键代码步骤。

**参考代码**:

```
// 简化版 shell 执行 cat file.txt | wc -l 的逻辑
int pid1, pid2;
int fd[2];
pipe(fd);

if ((pid1 = fork()) == 0) {
    // 子进程1：执行 cat
    close(fd[0]);          // 关闭读端
    dup2(fd[1], STDOUT_FILENO);  // 重定向 stdout 到 pipe 写端
    close(fd[1]);          // 关闭原始写端（为什么？）
    execlp("cat", "cat", "file.txt", NULL);
}

if ((pid2 = fork()) == 0) {
    // 子进程2：执行 wc
    close(fd[1]);          // 关闭写端
    dup2(fd[0], STDIN_FILENO);   // 重定向 stdin 到 pipe 读端
    close(fd[0]);          // 关闭原始读端
    execlp("wc", "wc", "-l", NULL);
}

// 父进程 shell
close(fd[0]);
close(fd[1]);
waitpid(pid1, NULL, 0);
waitpid(pid2, NULL, 0);
```
